cmake_minimum_required(VERSION 3.16)
project(cross_language VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cross_core src/space.cpp src/language.cpp src/machine.cpp src/structural.cpp src/compact.cpp src/grid.cpp)
target_include_directories(cross_core PUBLIC src)
target_compile_options(cross_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(cross src/main.cpp)
target_link_libraries(cross PRIVATE cross_core)
add_executable(cross_tests tests/test.cpp)
target_link_libraries(cross_tests PRIVATE cross_core)
target_compile_definitions(cross_tests PRIVATE EXAMPLE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
enable_testing()
add_test(NAME cross_semantics COMMAND cross_tests)
add_test(NAME cross_cli_resume COMMAND ${CMAKE_COMMAND} -DCROSS=$<TARGET_FILE:cross> -DEXAMPLES=${CMAKE_CURRENT_SOURCE_DIR}/examples -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/cli-test -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
add_executable(cross_bench tests/benchmark.cpp)
target_link_libraries(cross_bench PRIVATE cross_core)
add_executable(cross_audit tests/audit.cpp)
target_link_libraries(cross_audit PRIVATE cross_core)
target_compile_definitions(cross_audit PRIVATE EXAMPLE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
add_test(NAME cross_audit COMMAND cross_audit)
set_tests_properties(cross_semantics cross_cli_resume cross_audit PROPERTIES TIMEOUT 120)
add_executable(cross_structure tests/structure.cpp)
target_link_libraries(cross_structure PRIVATE cross_core)
target_compile_definitions(cross_structure PRIVATE EXAMPLE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
add_test(NAME cross_structure COMMAND cross_structure)
set_tests_properties(cross_structure PROPERTIES TIMEOUT 120)
add_executable(cross_grid tests/grid.cpp)
target_link_libraries(cross_grid PRIVATE cross_core)
target_compile_definitions(cross_grid PRIVATE EXAMPLE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
add_test(NAME cross_grid COMMAND cross_grid)
set_tests_properties(cross_grid PROPERTIES TIMEOUT 120)
add_executable(cross_chunks tests/chunks.cpp)
target_link_libraries(cross_chunks PRIVATE cross_core)
target_compile_definitions(cross_chunks PRIVATE EXAMPLE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
add_test(NAME cross_chunks COMMAND cross_chunks)
set_tests_properties(cross_chunks PROPERTIES TIMEOUT 120)
